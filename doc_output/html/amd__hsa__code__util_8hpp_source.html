150 // Helper function that allocates an aligned memory.
151 inline void*
152 alignedMalloc(size_t size, size_t alignment)
153 {
154 #if defined(_WIN32)
155  return ::_aligned_malloc(size, alignment);
156 #else
157  void * ptr = NULL;
158  alignment = (std::max)(alignment, sizeof(void*));
159  if (0 == ::posix_memalign(&ptr, alignment, size)) {
160  return ptr;
161  }
162  return NULL;
163 #endif
164 }

177 inline uint64_t alignUp(uint64_t num, uint64_t align)
178 {
179  assert(align);
180  assert((align & (align - 1)) == 0);
181  return (num + align - 1) & ~(align - 1);
182 }
