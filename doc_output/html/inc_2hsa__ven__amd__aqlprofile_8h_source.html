62 // Library API:
63 // The library provides helper methods for instantiation of
64 // the profile context object and for populating of the start
65 // and stop AQL packets. The profile object contains a profiling
66 // events list and needed for profiling buffers descriptors,
67 // a command buffer and an output data buffer. To check if there
68 // was an error the library methods return a status code. Also
69 // the library provides methods for querying required buffers
70 // attributes, to validate the event attributes and to get profiling
71 // output data.
72 //
73 // Returned status:
74 // hsa_status_t – HSA status codes are used from hsa.h header
75 //
76 // Supported profiling features:
77 //
78 // Supported profiling events

165 //
166 // Profile context object:
167 // The library provides a profile object structure which contains
168 // the events array, a buffer for the profiling start/stop commands
169 // and a buffer for the output data.
170 // The buffers are specified by the buffer descriptors and allocated
171 // by the application. The buffers allocation attributes, the command
172 // buffer size, the PMC output buffer size as well as profiling output
173 // data can be get using the generic get profile info helper _get_info.
174 //
175 // Buffer descriptor
176 typedef struct {
177  void* ptr;
178  uint32_t size;
179 } hsa_ven_amd_aqlprofile_descriptor_t;
180 
181 // Profile context object structure, contains profiling events list and
182 // needed for profiling buffers descriptors, a command buffer and
183 // an output data buffer
184 typedef struct {
185  hsa_agent_t agent; // GFXIP handle
186  hsa_ven_amd_aqlprofile_event_type_t type; // Events type
187  const hsa_ven_amd_aqlprofile_event_t* events; // Events array
188  uint32_t event_count; // Events count
189  const hsa_ven_amd_aqlprofile_parameter_t* parameters; // Parameters array
190  uint32_t parameter_count; // Parameters count
191  hsa_ven_amd_aqlprofile_descriptor_t output_buffer; // Output buffer
192  hsa_ven_amd_aqlprofile_descriptor_t command_buffer; // PM4 commands
193 } hsa_ven_amd_aqlprofile_profile_t;
