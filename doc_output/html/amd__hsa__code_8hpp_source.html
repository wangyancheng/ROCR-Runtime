77 template<typename class_type, typename member_type>
78 size_t OffsetOf(member_type class_type::*member)
79 {
80  return (char*)&((class_type*)nullptr->*member) - (char*)nullptr;
81 }
82 
83 template<typename class_type>
84 class_type* ObjectAt(uint64_t address)
85 {
86  if (!IsAccessibleMemoryAddress(address)) {
87  return nullptr;
88  }
89 
90  const uint64_t *rt_signature =
91  (const uint64_t*)(address + OffsetOf(&class_type::RT_SIGNATURE));
92  if (nullptr == rt_signature) {
93  return nullptr;
94  }
95  if (class_type::CT_SIGNATURE != *rt_signature) {
96  return nullptr;
97  }
98 
99  return (class_type*)address;
100 }
