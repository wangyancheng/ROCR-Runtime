56 class ScratchCache {
57  public:
58  struct node {
59  enum STATE { FREE = 0, ALLOC = 1, TRIM = 2, STEAL = 4 };
60  void* base;
61  bool large;
62  uint32_t state;
63 
64  node() : base(nullptr), state(FREE) {}
65 
66  bool isFree() const { return state == FREE; }
67  bool trimPending() const { return state == (ALLOC | TRIM); }
68 
69  void trim() {
70  assert(!isFree() && "Trim of free scratch node.");
71  state |= TRIM;
72  }
73  void free() {
74  assert(!isFree() && "Free of free scratch node.");
75  state = FREE;
76  }
77  void alloc() {
78  assert(isFree() && "Alloc of non-free scratch node.");
79  state = ALLOC;
80  }
81  };

116  bool alloc(ScratchInfo& info) {
117  ref_t it = map.upper_bound(info.size - 1);
118  if (it == map.end()) return false;
119 
120  // Small requests must have an exact size match and be small.
121  if (!info.large) {
122  while ((it != map.end()) && (it->first == info.size)) {
123  if (it->second.isFree() && (!it->second.large)) {
124  it->second.alloc();
125  info.queue_base = it->second.base;
126  info.scratch_node = it;
127  available_bytes_ -= it->first;
128  return true;
129  }
130  it++;
131  }
132  return false;
133  }
134 
135  // Large requests may use a small allocation and do not require an exact size match.
136  while (it != map.end()) {
137  if (it->second.isFree()) {
138  it->second.alloc();
139  info.queue_base = it->second.base;
140  info.scratch_node = it;
141  available_bytes_ -= it->first;
142  return true;
143  }
144  it++;
145  }
146  return false;
147  }
148 
149  void free(ScratchInfo& info) {
150  if (info.scratch_node == map.end()) {
151  // This is reserved scratch memory. Do not de-allocate, just mark it as free.
152  assert(!reserved_.second.isFree() && "free called when reserved node already free.");
153  reserved_.second.free();
154  available_bytes_ += reserved_.first;
155  return;
156  }
157 
158  assert(!info.scratch_node->second.isFree() && "free called on free scratch node.");
159  auto it = info.scratch_node;
160  if (it->second.trimPending()) {
161  dealloc(it->second.base, it->first, it->second.large);
162  map.erase(it);
163  return;
164  }
165  it->second.free();
166  available_bytes_ += it->first;
167  }
168 
169  bool trim(bool trim_nodes_in_use) {
170  bool ret = !map.empty();
171  auto it = map.begin();
172  while (it != map.end()) {
173  if (it->second.isFree()) {
174  available_bytes_ -= it->first;
175  dealloc(it->second.base, it->first, it->second.large);
176  auto temp = it;
177  it++;
178  map.erase(temp);
179  } else {
180  if (trim_nodes_in_use) it->second.trim();
181  it++;
182  }
183  }
184  return ret;
185  }

187  void insert(ScratchInfo& info) {
188  node n;
189  n.base = info.queue_base;

191  n.alloc();
192 
193  auto it = map.insert(std::make_pair(info.size, n));
194  info.scratch_node = it;

200  void reserve(size_t bytes, void* base) {
201  assert(!reserved_.first && "Already reserved memory.");
202 
203  node n;
204  n.base = base;
205  n.large = 0;
206 
207  available_bytes_ += bytes;
208 
209  reserved_ = std::make_pair(bytes, n);
210  }
211 
212  bool use_reserved(ScratchInfo& info) {
213  if (!reserved_.second.isFree() || info.size > reserved_.first) {
214  debug_print("reserved node is already in use or too small (requested:%ld reserved:%ld)\n",
215  info.size, reserved_.first);
216  return false;
217  }
218  reserved_.second.large = info.large;
219  reserved_.second.alloc();
220  info.queue_base = reserved_.second.base;
221  // Special case to indicate that this node is reserved memory
222  info.scratch_node = map.end();
223  available_bytes_ -= reserved_.first;
224  return true;
225  }
226 
227  void free_reserve() {
228  available_bytes_ -= reserved_.first;
229  if (reserved_.first) dealloc(reserved_.second.base, reserved_.first, reserved_.second.large);
230 
231  reserved_.first = 0;
232  reserved_.second.base = NULL;
233  reserved_.second.large = 0;
234  }
