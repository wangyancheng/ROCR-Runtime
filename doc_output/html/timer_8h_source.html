55 // Needed to patch around a mixed arithmetic bug in MSVC's duration_cast as of
56 // VS 2013.
57 template <bool isFloat, bool isSigned>
58 struct wide_type {
59  typedef double type;
60 };
61 template <>

70 template <typename To, typename Rep, typename Period>
71 static __forceinline To
72  duration_cast(const std::chrono::duration<Rep, Period>& d) {
73  typedef typename wide_type<std::is_floating_point<Rep>::value,
74  std::is_signed<Rep>::value>::type wide;
75  typedef std::chrono::duration<wide, typename To::period> unit_convert_t;
76 
77  unit_convert_t temp = std::chrono::duration_cast<unit_convert_t>(d);
78  return To(static_cast<typename To::rep>(temp.count()));
79 }
