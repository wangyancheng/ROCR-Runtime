392 typedef std::pair<std::string, hsa_agent_t> AgentSymbol;
393 struct ASC {
394  bool operator()(const AgentSymbol &las, const AgentSymbol &ras) const {
395  return las.first == ras.first && las.second.handle == ras.second.handle;
396  }
397 };
398 struct ASH {
399  size_t operator()(const AgentSymbol &as) const {
400  size_t h = std::hash<std::string>()(as.first);
401  size_t i = std::hash<uint64_t>()(as.second.handle);
402  return h ^ (i << 1);
403  }
404 };
