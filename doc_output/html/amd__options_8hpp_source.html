228 template<typename T>
229 bool Option<T>::ProcessTokens(std::list<std::string> &tokens) {
230  assert(0 == name_.compare(tokens.front()) && "option name is mismatched");
231  if (2 > tokens.size()) {
232  error() << "error: invalid option: \'" << name_ << '\'' << std::endl;
233  return false;
234  }
235 
236  is_set_ = true;
237  tokens.pop_front();
238 
239  while (!tokens.empty()) {
240  std::istringstream token_stream(tokens.front());
241  if (!token_stream.good()) {
242  error() << "error: invalid option: \'" << name_ << '\'' << std::endl;
243  return false;
244  }
245 
246  T value;
247  token_stream >> value;
248 
249  values_.push_back(value);
250  tokens.pop_front();
251  }
252  return true;
253 }

374 
375  bool ProcessTokens(std::list<std::string> &tokens) override {
376  assert(0 == name_.compare(tokens.front()) && "option name is mismatched");
377  if (1 == tokens.size()) {
378  tokens.pop_front();
379  is_set_ = true;
380  return true;
381  } else if (2 == tokens.size()) {
382  tokens.pop_front();
383  if (tokens.front() == "1") {
384  is_set_ = true;
385  tokens.pop_front();
386  return true;
387  } else if (tokens.front() == "0") {
388  is_set_ = false;
389  tokens.pop_front();
390  return true;
391  }
392  }
393  error() << "error: invalid option: '" << name_ << "'" << std::endl;
394  return false;
395  }
