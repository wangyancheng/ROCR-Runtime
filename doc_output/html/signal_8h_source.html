103  void GetSdmaTsAddresses(uint64_t*& start, uint64_t*& end) {
104  /*
105  SDMA timestamps on gfx7xx/8xxx require 32 byte alignment (gfx9xx relaxes
106  alignment to 8 bytes). This conflicts with the frozen format for amd_signal_t
107  so we place the time stamps in sdma_start/end_ts instead (amd_signal.start_ts
108  is also properly aligned). Reading of the timestamps occurs in GetRawTs().
109  */
110  start = &sdma_start_ts;
111  end = &sdma_end_ts;
112  }
113 
114  void CopyPrep() {
115  // Clear sdma_end_ts before a copy so we can detect if the copy was done via
116  // SDMA or blit kernel.
117  sdma_start_ts = 0;
118  sdma_end_ts = 0;
119  }
120 
121  void GetRawTs(bool FetchCopyTs, uint64_t& start, uint64_t& end) {
122  /*
123  If the read is for a copy we need to check if it was done by blit kernel or SDMA.
124  Since we clear sdma_start/end_ts during CopyPrep we know it was a SDMA copy if one
125  of those is non-zero. Otherwise return compute kernel stamps from amd_signal.
126  */
127  if (FetchCopyTs && sdma_end_ts != 0) {
128  start = sdma_start_ts;
129  end = sdma_end_ts;
130  return;
131  }
132  start = amd_signal.start_ts;
133  end = amd_signal.end_ts;
134  }

244  static __forceinline Signal* Convert(hsa_signal_t signal) {
245  if (signal.handle == 0) throw AMD::hsa_exception(HSA_STATUS_ERROR_INVALID_ARGUMENT, "");
246  SharedSignal* shared = SharedSignal::Convert(signal);
247  if (!shared->IsValid())
248  throw AMD::hsa_exception(HSA_STATUS_ERROR_INVALID_SIGNAL, "Signal handle is invalid.");
249  if (shared->IsIPC()) {
250  Signal* ret = lookupIpc(signal);
251  if (ret == nullptr)
252  throw AMD::hsa_exception(HSA_STATUS_ERROR_INVALID_SIGNAL, "Signal handle is invalid.");
253  return ret;
254  } else {
255  return shared->core_signal;
256  }
257  }

259  static Signal* DuplicateHandle(hsa_signal_t signal) {
260  if (signal.handle == 0) return nullptr;
261  SharedSignal* shared = SharedSignal::Convert(signal);
262 
263  if (!shared->IsIPC()) {
264  if (!shared->IsValid()) return nullptr;
265  shared->core_signal->refcount_++;
266  shared->core_signal->Retain();
267  return shared->core_signal;
268  }
269 
270  // IPC signals may only be duplicated while holding the ipcMap lock.
271  return duplicateIpc(signal);
272  }
