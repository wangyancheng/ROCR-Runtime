73 template <typename T> class PageAllocator : private BaseShared {
74  public:
75  __forceinline static T* alloc(int flags = 0) {
76  T* ret = reinterpret_cast<T*>(allocate_(AlignUp(sizeof(T), 4096), 4096, flags));
77  if (ret == nullptr) throw std::bad_alloc();
78 
79  MAKE_NAMED_SCOPE_GUARD(throwGuard, [&]() { free_(ret); });
80 
81  new (ret) T;
82 
83  throwGuard.Dismiss();
84  return ret;
85  }
86 
87  __forceinline static void free(T* ptr) {
88  if (ptr != nullptr) {
89  ptr->~T();
90  free_(ptr);
91  }
92  }
93 };

97 template <typename T, typename Allocator = PageAllocator<T>>
98 class Shared final : private BaseShared {
99  public:
100  explicit Shared(Allocator* pool = nullptr, int flags = 0) : pool_(pool) {
101  assert(allocate_ != nullptr && free_ != nullptr &&
102  "Shared object allocator is not set");
103 
104  if (pool_)
105  shared_object_ = pool_->alloc();
106  else
107  shared_object_ = PageAllocator<T>::alloc(flags);
108  }
109 
110  ~Shared() {
111  assert(allocate_ != nullptr && free_ != nullptr && "Shared object allocator is not set");
112 
113  if (pool_)
114  pool_->free(shared_object_);

177 template <typename T, size_t Align> class SharedArray final : private BaseShared {
178  public:
179  SharedArray() : shared_object_(nullptr) {}
180 
181  explicit SharedArray(size_t length) : shared_object_(nullptr), len(length) {
182  assert(allocate_ != nullptr && free_ != nullptr && "Shared object allocator is not set");
183  static_assert((__alignof(T) <= Align) || (Align == 0), "Align is less than alignof(T)");
184 
185  shared_object_ =
186  reinterpret_cast<T*>(allocate_(sizeof(T) * length, Max(__alignof(T), Align), 0));
187  if (shared_object_ == nullptr) throw std::bad_alloc();
188 
189  size_t i = 0;
190 
191  MAKE_NAMED_SCOPE_GUARD(loopGuard, [&]() {
192  for (size_t t = 0; t < i - 1; t++) shared_object_[t].~T();
193  free_(shared_object_);
194  });
195 
196  for (; i < length; i++) new (&shared_object_[i]) T;
197 
198  loopGuard.Dismiss();
199  }
200 
201  ~SharedArray() {
202  assert(allocate_ != nullptr && free_ != nullptr && "Shared object allocator is not set");
203 
204  if (shared_object_ != nullptr) {
205  for (size_t i = 0; i < len; i++) shared_object_[i].~T();
206  free_(shared_object_);
207  }
208  }
209 
210  SharedArray(SharedArray&& rhs) {
211  this->~SharedArray();
212  shared_object_ = rhs.shared_object_;
213  rhs.shared_object_ = nullptr;
214  len = rhs.len;
215  }
216  SharedArray& operator=(SharedArray&& rhs) {
217  this->~SharedArray();
218  shared_object_ = rhs.shared_object_;
219  rhs.shared_object_ = nullptr;
220  len = rhs.len;
221  return *this;
222  }
223 
224  T& operator[](size_t index) {
225  assert(index < len && "Index out of bounds.");
226  return shared_object_[index];
227  }
228  const T& operator[](size_t index) const {
229  assert(index < len && "Index out of bounds.");
230  return shared_object_[index];
231  }
232 
233  private:
234  T* shared_object_;
235  size_t len;
