147 class ScopeGuard {
148  public:
149  explicit __forceinline ScopeGuard(const lambda& release)
150  : release_(release), dismiss_(false) {}
151 
152  ScopeGuard(ScopeGuard& rhs) { *this = rhs; }
153 
154  __forceinline ~ScopeGuard() {
155  if (!dismiss_) release_();
156  }
157  __forceinline ScopeGuard& operator=(ScopeGuard& rhs) {
158  dismiss_ = rhs.dismiss_;
159  release_ = rhs.release_;
160  rhs.dismiss_ = true;
161  return *this;
162  }
163  __forceinline void Dismiss() { dismiss_ = true; }
164 
165  private:

167  bool dismiss_;

294 static __forceinline uint32_t NextPow2(uint32_t value) {
295  if (value == 0) return 1;

337 template <uint32_t lowBit, uint32_t highBit, typename T>
338 static __forceinline uint32_t BitSelect(T p) {
339  static_assert(sizeof(T) <= sizeof(uintptr_t), "Type out of range.");
340  static_assert(highBit < sizeof(uintptr_t) * 8, "Bit index out of range.");
341 
342  uintptr_t ptr = p;
343  if (highBit != (sizeof(uintptr_t) * 8 - 1))
344  return (uint32_t)((ptr & ((1ull << (highBit + 1)) - 1)) >> lowBit);
345  else
346  return (uint32_t)(ptr >> lowBit);
347 }
