55 /*
56  * Wrapper for a std::unique_ptr that initializes its object at first use.
57  */
58 template <typename T> class lazy_ptr {
59  public:
60  lazy_ptr() {}
61 
62  explicit lazy_ptr(std::function<T*()> Constructor) { Init(Constructor); }
63 
64  lazy_ptr(lazy_ptr&& rhs) {
65  obj = std::move(rhs.obj);
66  func = std::move(rhs.func);
67  }
68 
69  lazy_ptr& operator=(lazy_ptr&& rhs) {
70  obj = std::move(rhs.obj);
71  func = std::move(rhs.func);
72  }
73

75  lazy_ptr& operator=(lazy_ptr&) = delete;

78  obj.reset();

80  }

106  /*
107  * Ensures that the object is created or is being created.
108  * This is useful when early construction of the object is required.
109  */
110  void touch() const { make(false); }
111 
112  // Tells if the lazy object has been constructed or not.
113  // Construction may fail silently (return nullptr).
114  bool created() const {
115  std::atomic_thread_fence(std::memory_order_acquire);
116  return func == nullptr;
117  }
118 
119  // Tells if the lazy object exists or not.
120  bool empty() const {
121  std::atomic_thread_fence(std::memory_order_acquire);
122  return obj == nullptr;
123  }
124 
125  private:
126  mutable std::unique_ptr<T> obj;
127  mutable std::function<T*(void)> func;
128  mutable KernelMutex lock;
129 
130  // Separated from make to improve inlining.
131  void make_body(bool block) const {
132  if (block) {
133  lock.Acquire();
134  } else if (!lock.Try()) {
135  return;
136  }
137  MAKE_SCOPE_GUARD([&]() { lock.Release(); });
138  if (func == nullptr) return;
139  T* ptr = func();
140  obj.reset(ptr);
141  std::atomic_thread_fence(std::memory_order_release);
142  func = nullptr;
143  }
144 
145  __forceinline void make(bool block) const {
146  if (!created()) {
147  make_body(block);
148  }
149  }
150 
151 };
