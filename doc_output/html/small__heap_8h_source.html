92  public:
93  SmallHeap() : pool(nullptr), length(0), total_free(0) {}
94  SmallHeap(void* base, size_t length)
95  : pool(base), length(length), total_free(length) {
96  assert(pool != nullptr && "Invalid base address.");
97  assert(pool != (void*)0xFFFFFFFFFFFFFFFFull && "Invalid base address.");
98  assert((char*)pool + length != (char*)0xFFFFFFFFFFFFFFFFull && "Invalid pool bounds.");
99 
100  Node& start = memory[0];
101  Node& node = memory[pool];
102  Node& end = memory[(void*)0xFFFFFFFFFFFFFFFFull];
103 
104  start.len = 0;
105  start.next = memory.find(pool);
106  setfirstfree(start);
107 
108  node.len = length;
109  node.prior = memory.begin();
110  node.next = --memory.end();
111 
112  end.len = 0;
113  end.prior = start.next;
114  setlastfree(end);
115 
116  high.insert((void*)0xFFFFFFFFFFFFFFFFull);
117  }
