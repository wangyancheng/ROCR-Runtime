295 template<uint32_t lowBit, uint32_t highBit, typename T>
296 static __forceinline uint32_t BitSelect(T p) {
297  static_assert(sizeof(T) <= sizeof(uintptr_t), "Type out of range.");
298  static_assert(highBit < sizeof(uintptr_t)*8, "Bit index out of range.");
299 
300  uintptr_t ptr = p;
301  if(highBit != (sizeof(uintptr_t)*8-1))
302  return (uint32_t)((ptr & ((1ull<<(highBit+1))-1)) >> lowBit);
303  else
304  return (uint32_t)(ptr >> lowBit);
305 }
