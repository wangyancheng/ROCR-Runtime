43 // A simple best fit memory allocator with eager compaction. Manages block sub-allocation.
44 // For use when memory efficiency is more important than allocation speed.
45 // O(log n) time.

64  size_t size : 62;

66  bool free : 1;

109  if (node.free_list_entry_ != free_list_.end()) {

133  void* alloc(size_t bytes) {
134  // Find best fit.
135  auto free_fragment = free_list_.lower_bound(bytes);
136  uintptr_t base;
137  size_t size;
138 
139  if (free_fragment != free_list_.end()) {
140  base = free_fragment->second;
141  size = free_fragment->first;
142  free_list_.erase(free_fragment);
143 
144  assert(size >= bytes && "SimpleHeap: map lower_bound failure.");
145 
146  // Find the containing block and fragment
147  auto it = block_list_.upper_bound(base);
148  it--;
149  auto& frag_map = it->second;
150  const auto& fragment = frag_map.find(base);
151 
152  assert(fragment != frag_map.end() && "Inconsistency in SimpleHeap.");
153  assert(size == fragment->second.size && "Inconsistency in SimpleHeap.");
154 
155  // Sub-allocate from fragment.
156  fragment->second.size = bytes;
157  setUsed(fragment->second);
158  // Record remaining free space.
159  if (size > bytes) {
160  free_fragment = free_list_.insert(std::make_pair(size - bytes, base + bytes));
161  frag_map[base + bytes] = makeFragment(free_fragment, size - bytes);
162  }
163  return reinterpret_cast<void*>(base);
164  }
165 
166  // No usable fragment, check block cache
167  if (bytes < default_block_size() && !block_cache_.empty()) {
168  const auto& block = block_cache_.back();
169  base = block.base_ptr_;
170  size = block.length_;
171  block_cache_.pop_back();
172  cache_size_ -= size;
173  } else { // Alloc new block - new block may be larger than default.
174  void* ptr = block_allocator_.alloc(bytes, size);
175  base = reinterpret_cast<uintptr_t>(ptr);
176  assert(ptr != nullptr && "Block allocation failed, Allocator is expected to throw.");
177  }
178 
179  in_use_size_ += size;
180  assert(size >= bytes && "Alloc exceeds block size.");
181  // Sub alloc and insert free region.
182  if (size > bytes) {
183  free_fragment = free_list_.insert(std::make_pair(size - bytes, base + bytes));
184  block_list_[base][base + bytes] = makeFragment(free_fragment, size - bytes);
185  }
186  // Track used region
187  block_list_[base][base] = makeFragment(bytes);
188 
189  // Disallow multiple suballocation from large blocks.
190  // Prevents a small allocation from retaining a large block.
191  if (bytes > default_block_size()) {
192  bool err = discardBlock(reinterpret_cast<void*>(base));
193  assert(err && "Large block discard failed.");
194  }
195 
196  return reinterpret_cast<void*>(base);
197  }
198 
199  bool free(void* ptr) {
200  if (ptr == nullptr) return true;
201 
202  uintptr_t base = reinterpret_cast<uintptr_t>(ptr);
203 
204  // Find fragment and validate.
205  auto frag_map_it = block_list_.upper_bound(base);
206  if (frag_map_it == block_list_.begin()) return false;
207  frag_map_it--;
208  auto& frag_map = frag_map_it->second;
209  auto fragment = frag_map.find(base);
210  if (fragment == frag_map.end() || isFree(fragment->second)) return false;
211 
212  bool discard = fragment->second.discard;
213 
214  // Merge lower
215  if (fragment != frag_map.begin()) {
216  auto lower = fragment;
217  lower--;
218  if (isFree(lower->second)) {
219  removeFreeListEntry(lower->second);
220  lower->second.size += fragment->second.size;
221  frag_map.erase(fragment);
222  fragment = lower;
223  }
224  }
225 
226  // Merge upper
227  {
228  auto upper = fragment;
229  upper++;
230  if ((upper != frag_map.end()) && isFree(upper->second)) {
231  removeFreeListEntry(upper->second);
232  fragment->second.size += upper->second.size;
233  frag_map.erase(upper);
234  }
235  }
236 
237  // Release whole free blocks.
238  if (frag_map.size() == 1) {
239  Block block(fragment->first, fragment->second.size);
240  block_list_.erase(frag_map_it);
241 
242  // Discard or add to the block cache.
243  if (discard) {
244  block_allocator_.free(reinterpret_cast<void*>(block.base_ptr_), block.length_);
245  } else {
246  block_cache_.push_back(block);
247  cache_size_ += block.length_;
248  in_use_size_ -= block.length_;
249  }
250 
251  balance();
252 
253  // Don't publish free space since block was moved to the cache.
254  return true;
255  }
256 
257  // Don't report free memory if discarding the fragment.
258  if (discard) return true;
259 
260  // Report free fragment
261  const auto& freeEntry =
262  free_list_.insert(std::make_pair(size_t(fragment->second.size), fragment->first));
263  setFree(fragment->second, freeEntry);
264 
265  return true;
266  }
267 
268  void balance() {
269  // Release old blocks when over cache limit.
270  while ((block_cache_.size() > 1) && (cache_size_ > in_use_size_ * 2)) {
271  const auto& block = block_cache_.front();
272  block_allocator_.free(reinterpret_cast<void*>(block.base_ptr_), block.length_);
273  cache_size_ -= block.length_;
274  block_cache_.pop_front();
275  }
276  }
277 
278  void trim() {
279  for (const auto& block : block_cache_)
280  block_allocator_.free(reinterpret_cast<void*>(block.base_ptr_), block.length_);
281  block_cache_.clear();
282  cache_size_ = 0;
283  }

285  size_t cache_size() const { return cache_size_; }
286 
287  size_t default_block_size() const { return block_allocator_.block_size(); }
288 
289  // Prevent reuse of the block containing ptr. No further fragments will be allocated from the
290  // block and the block will not be added to the block cache when it is free.
291  bool discardBlock(void* ptr) {
292  if (ptr == nullptr) return true;
293 
294  uintptr_t base = reinterpret_cast<uintptr_t>(ptr);
295 
296  // Find block validate.
297  auto frag_map_it = block_list_.upper_bound(base);
298  if (frag_map_it == block_list_.begin()) return false;
299  frag_map_it--;
300  auto& frag_map = frag_map_it->second;
301  if ((base < frag_map.begin()->first) ||
302  (frag_map.rbegin()->first + frag_map.rbegin()->second.size <= base))
303  return false;
304 
305  // Is block already discarded?
306  if (frag_map.begin()->second.discard) return true;
307 
308  // Mark all fragments for discard and compute block size. Removes freelist records for all
309  // fragments in the block.
310  size_t size = 0;
311  for (auto& frag : frag_map) {
312  discard(frag.second);
313  size += frag.second.size;
314  }
315 
316  // Remove discarded block from in-use tracking and rebalance the block cache.
317  in_use_size_ -= size;
318  balance();
319 
320  return true;
321  }
322 };
