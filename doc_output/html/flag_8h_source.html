97

99  max_queues_ = static_cast<uint32_t>(atoi(var.c_str()));
100 
101  // Maximum amount of scratch mem that can be used per process per gpu
102  var = os::GetEnvVar("HSA_SCRATCH_MEM");
103  scratch_mem_size_ = atoi(var.c_str());
104 
105  // Scratch memory sizes > HSA_SCRATCH_SINGLE_LIMIT will trigger a use-once scheme
106  // We also reserve HSA_SCRATCH_SINGLE_LIMIT per process per gpu to guarrantee we
107  // have sufficient memory to for scratch in case user tried to allocate all device
108  // memory
109  if (os::IsEnvVarSet("HSA_SCRATCH_SINGLE_LIMIT")) {
110  var = os::GetEnvVar("HSA_SCRATCH_SINGLE_LIMIT");
111  scratch_single_limit_ = atoi(var.c_str());
112  } else {
113  scratch_single_limit_ = DEFAULT_SCRATCH_SINGLE_LIMIT;
114  }
115 
116  tools_lib_names_ = os::GetEnvVar("HSA_TOOLS_LIB");
117 
118  var = os::GetEnvVar("HSA_TOOLS_REPORT_LOAD_FAILURE");
119 
120  ifdebug {

124  }

131 
132  var = os::GetEnvVar("HSA_REV_COPY_DIR");
133  rev_copy_dir_ = (var == "1") ? true : false;
134 
135  var = os::GetEnvVar("HSA_FORCE_FINE_GRAIN_PCIE");
136  fine_grain_pcie_ = (var == "1") ? true : false;
137 
138  var = os::GetEnvVar("HSA_NO_SCRATCH_RECLAIM");
139  no_scratch_reclaim_ = (var == "1") ? true : false;
140 
141  var = os::GetEnvVar("HSA_NO_SCRATCH_THREAD_LIMITER");
142  no_scratch_thread_limit_ = (var == "1") ? true : false;
143 
144  var = os::GetEnvVar("HSA_DISABLE_IMAGE");
145  disable_image_ = (var == "1") ? true : false;
146 
147  var = os::GetEnvVar("HSA_LOADER_ENABLE_MMAP_URI");
148  loader_enable_mmap_uri_ = (var == "1") ? true : false;
149 
150  var = os::GetEnvVar("HSA_FORCE_SDMA_SIZE");
151  force_sdma_size_ = var.empty() ? 1024 * 1024 : atoi(var.c_str());
152 
153  var = os::GetEnvVar("HSA_IGNORE_SRAMECC_MISREPORT");
154  check_sramecc_validity_ = (var == "1") ? false : true;
155 
156  // Legal values are zero "0" or one "1". Any other value will
157  // be interpreted as not defining the env variable.
158  var = os::GetEnvVar("HSA_XNACK");
159  xnack_ = (var == "0") ? XNACK_DISABLE : ((var == "1") ? XNACK_ENABLE : XNACK_UNCHANGED);
160 
161  var = os::GetEnvVar("HSA_ENABLE_DEBUG");
162  debug_ = (var == "1") ? true : false;
163 
164  var = os::GetEnvVar("HSA_CU_MASK_SKIP_INIT");
165  cu_mask_skip_init_ = (var == "1") ? true : false;
166 
167  // Temporary opt-in for corrected HSA_AMD_AGENT_INFO_COOPERATIVE_COMPUTE_UNIT_COUNT behavior.
168  // Will become opt-out and possibly removed in future releases.
169  var = os::GetEnvVar("HSA_COOP_CU_COUNT");
170  coop_cu_count_ = (var == "1") ? true : false;
171 
172  var = os::GetEnvVar("HSA_DISCOVER_COPY_AGENTS");
173  discover_copy_agents_ = (var == "1") ? true : false;
174 
175  var = os::GetEnvVar("HSA_SVM_PROFILE");
176  svm_profile_ = var;
177 
178  var = os::GetEnvVar("HSA_IMAGE_PRINT_SRD");
179  image_print_srd_ = (var == "1") ? true : false;
180 
181  var = os::GetEnvVar("HSA_ENABLE_MWAITX");
182  enable_mwaitx_ = (var == "1") ? true : false;
183 
184  // Temporary environment variable to disable CPU affinity override
185  // Will either rename to HSA_OVERRIDE_CPU_AFFINITY later or remove completely.
186  var = os::GetEnvVar("HSA_OVERRIDE_CPU_AFFINITY_DEBUG");
187  override_cpu_affinity_ = (var == "0") ? false : true;
188  }
