54 // Creates enumeration entries used for packing bits into integers. Enumeration
55 // entries include bit shift amount, bit width, and bit mask.
56 #ifndef AMDHSA_BITS_ENUM_ENTRY
57 #define AMDHSA_BITS_ENUM_ENTRY(NAME, SHIFT, WIDTH) \
58  NAME ## _SHIFT = (SHIFT), \
59  NAME ## _WIDTH = (WIDTH), \
60  NAME = (((1 << (WIDTH)) - 1) << (SHIFT))
61 #endif // AMDHSA_BITS_ENUM_ENTRY
62 
63 // Gets bits for specified bit mask from specified source.
64 #ifndef AMDHSA_BITS_GET
65 #define AMDHSA_BITS_GET(SRC, MSK) ((SRC & MSK) >> MSK ## _SHIFT)
66 #endif // AMDHSA_BITS_GET
67 
68 // Sets bits for specified bit mask in specified destination.
69 #ifndef AMDHSA_BITS_SET
70 #define AMDHSA_BITS_SET(DST, MSK, VAL) \
71  DST &= ~MSK; \
72  DST |= ((VAL << MSK ## _SHIFT) & MSK)
73 #endif // AMDHSA_BITS_SET
