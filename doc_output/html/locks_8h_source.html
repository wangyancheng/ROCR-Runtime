76 class SpinMutex {
77  public:
78  SpinMutex() { lock_ = 0; }
79 
80  bool Try() {
81  int old = 0;
82  return lock_.compare_exchange_strong(old, 1);
83  }
84  bool Acquire() {
85  int old = 0;
86  while (!lock_.compare_exchange_strong(old, 1))
87  {
88  old=0;
89  os::YieldThread();
90  }
91  return true;
92  }
93  void Release() { lock_ = 0; }
94 
95  private:

179 template <class LockType> class ScopedAcquire {
180  public:
183  explicit ScopedAcquire(LockType* lock) : lock_(lock), doRelease(true) {
184  static_assert(isMutex<LockType>::value, "ScopedAcquire requires a mutex type.");
185  lock_.Acquire();
186  }
187  explicit ScopedAcquire(LockType lock) : lock_(lock), doRelease(true) {
188  static_assert(!isMutex<LockType>::value, "Mutex types are not copyable.");
189  lock_.Acquire();
190  }
191 
193  ~ScopedAcquire() {
194  if (doRelease) lock_.Release();
195  }
196 
198  void Release() {
199  lock_.Release();
200  doRelease = false;
201  }
202 
203  private:
