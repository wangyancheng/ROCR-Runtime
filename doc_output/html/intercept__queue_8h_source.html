135 // @brief Generic container for a proxy queue.
136 // Presents an proxy packet buffer and doorbell signal for an underlying Queue. Write index
137 // operations act on the proxy buffer while all other operations pass through to the underlying
138 // queue.
139 class QueueProxy : public QueueWrapper {
140  public:
141  explicit QueueProxy(std::unique_ptr<Queue> queue) : QueueWrapper(std::move(queue)) {}
142 
143  uint64_t LoadReadIndexAcquire() override {
144  return atomic::Load(&amd_queue_.read_dispatch_id, std::memory_order_acquire);
145  }
146  uint64_t LoadReadIndexRelaxed() override {
147  return atomic::Load(&amd_queue_.read_dispatch_id, std::memory_order_relaxed);
148  }
149  void StoreReadIndexRelaxed(uint64_t value) override { assert(false); }
150  void StoreReadIndexRelease(uint64_t value) override { assert(false); }
151 
152  uint64_t LoadWriteIndexRelaxed() override {
153  return atomic::Load(&amd_queue_.write_dispatch_id, std::memory_order_relaxed);
154  }
155  uint64_t LoadWriteIndexAcquire() override {
156  return atomic::Load(&amd_queue_.write_dispatch_id, std::memory_order_acquire);
157  }
158  void StoreWriteIndexRelaxed(uint64_t value) override {
159  atomic::Store(&amd_queue_.write_dispatch_id, value, std::memory_order_relaxed);
160  }
161  void StoreWriteIndexRelease(uint64_t value) override {
162  atomic::Store(&amd_queue_.write_dispatch_id, value, std::memory_order_release);
163  }
164  uint64_t CasWriteIndexAcqRel(uint64_t expected, uint64_t value) override {
165  return atomic::Cas(&amd_queue_.write_dispatch_id, value, expected, std::memory_order_acq_rel);
166  }
167  uint64_t CasWriteIndexAcquire(uint64_t expected, uint64_t value) override {
168  return atomic::Cas(&amd_queue_.write_dispatch_id, value, expected, std::memory_order_acquire);
169  }

171  return atomic::Cas(&amd_queue_.write_dispatch_id, value, expected, std::memory_order_relaxed);
172  }
173  uint64_t CasWriteIndexRelease(uint64_t expected, uint64_t value) override {
174  return atomic::Cas(&amd_queue_.write_dispatch_id, value, expected, std::memory_order_release);
175  }

186  return atomic::Add(&amd_queue_.write_dispatch_id, value, std::memory_order_release);
187  }

190 // @brief Provides packet intercept and rewrite capability for a queue.
191 // Host-side dispatches are processed during doorbell ring.
192 // Device-side dispatches are processed as an asynchronous signal event.
193 class InterceptQueue : public QueueProxy, private LocalSignal, public DoorbellSignal {
194  public:
195  explicit InterceptQueue(std::unique_ptr<Queue> queue);
196  ~InterceptQueue();
197 
198  void AddInterceptor(hsa_amd_queue_intercept_handler interceptor, void* data) {
199  assert(interceptor != nullptr && "Packet intercept callback was nullptr.");
200  interceptors.push_back(std::make_pair(interceptor, data));
201  }
202 
203  hsa_status_t Inactivate() override {
204  active_ = false;
205  return wrapped->Inactivate();
206  }
207 
208  private:
209  // Serialize packet interception processing.
210  KernelMutex lock_;
211 
212  // Largest processed packet index.
213  uint64_t next_packet_;
214 
215  // Post interception packet overflow buffer
216  std::vector<AqlPacket> overflow_;
217 
218  // Index at which async intercept processing was scheduled.
219  uint64_t retry_index_;
220 
221  // Event signal to use for async packet processing and control flag.
222  InterruptSignal* async_doorbell_;
223  std::atomic<bool> quit_;
224 
225  // Indicates queue active/inactive state.
226  std::atomic<bool> active_;
227 
228  // Proxy packet buffer
229  SharedArray<AqlPacket, 4096> buffer_;
230 
231  // Packet transform callbacks
232  std::vector<std::pair<AMD::callback_t<hsa_amd_queue_intercept_handler>, void*>> interceptors;
233 
234  static const hsa_signal_value_t DOORBELL_MAX = 0xFFFFFFFFFFFFFFFFull;
235 
236  static bool HandleAsyncDoorbell(hsa_signal_value_t value, void* arg);
237  static void PacketWriter(const void* pkts, uint64_t pkt_count);
238 
239  bool Submit(const AqlPacket* packets, uint64_t count);
240  static void Submit(const void* pkts, uint64_t pkt_count, uint64_t user_pkt_index, void* data,
241  hsa_amd_queue_intercept_packet_writer writer);
242 
243  /*
244  * Remaining Queue and Signal interface definitions.
245  */
246  public:
250  void StoreRelaxed(hsa_signal_value_t value) override;
251 
255  void StoreRelease(hsa_signal_value_t value) override {
256  std::atomic_thread_fence(std::memory_order_release);
257  StoreRelaxed(value);
258  }
259 
260  static __forceinline bool IsType(core::Signal* signal) { return signal->IsType(&rtti_id_); }

262 
263  protected:

265 
266  private:
267  static int rtti_id_;
